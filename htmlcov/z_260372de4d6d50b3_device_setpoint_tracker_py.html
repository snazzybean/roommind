1"""Setpoint changes made on a device itself (#259). 

2 

3RoomMind writes a setpoint to every device it controls and sends it again when 

4the device reports something else, so a turn of the TRV dial or a press on the 

5AC remote is undone within one cycle. Rooms that opt in take such a change over 

6as an override instead. 

7 

8State updates from Zigbee or cloud integrations carry no trace of who changed a 

9setpoint, not even the echo of RoomMind's own command. The tracker therefore 

10records every climate command RoomMind sends (``EVENT_CALL_SERVICE`` with the 

11RoomMind context) and judges by value: a reported setpoint that moved away from 

12the last one seen, is no value RoomMind sent and the device has not shown yet, 

13and did not come with a mode switch RoomMind sent was changed by someone else. 

14""" 

15

37def _as_float(value: Any) -> float | None: 

38 try: 

39 return float(value) if value is not None else None 

40 except (TypeError, ValueError): 

41 return None

44_ANY_ON = "*on" # turn_on: any mode but off

47def _mode_switched_to(service: str, data: Mapping[str, Any], current: str | None) -> str | None: 

48 """The hvac mode a command switches the device to, None if it keeps the current one.""" 

49 if service == "turn_off": 

50 return "off" if current != "off" else None 

51 if service == "turn_on": 

52 return _ANY_ON if current in (None, "off") else None 

53 mode = data.get("hvac_mode") 

54 return mode if mode is not None and mode != current else None

57def _shows(mode: str, hvac_state: str) -> bool: 

58 return hvac_state != "off" if mode == _ANY_ON else hvac_state == mode

65 self.at = at

70class DeviceSetpointTracker: 

71 """Tells setpoint changes made on a device apart from RoomMind's own commands.""" 

72 

73 def __init__(self, now: Callable[[], float]) -> None: 

74 self._now = now 

75 # {entity_id: last mode switch RoomMind sent} 

76 self._switch: dict[str, _ModeSwitch] = {} 

77 # {(entity_id, attr): [(time, value), ...]} setpoints RoomMind sent and the device has not shown yet 

78 self._sent: dict[tuple[str, str], list[tuple[float, float]]] = {} 

79 # {(entity_id, attr): (value reported at the last observation, its time)} 

80 self._seen: dict[tuple[str, str], tuple[float, float]] = {} 

81 # {entity_id: (hvac state at the last observation, when it was first seen)} 

82 self._state: dict[str, tuple[str, float]] = {} 

83 

84 def async_track(self, hass: HomeAssistant) -> Callable[[], None]: 

85 """Record every climate command RoomMind sends from now on; returns the unsubscribe.""" 

86 

87 @callback 

88 def _is_climate_call(event_data: Mapping[str, Any]) -> bool: 

89 return bool(event_data.get(ATTR_DOMAIN) == "climate") 

90 

91 @callback 

92 def _record(event: Event) -> None: 

93 if event.context.parent_id != ROOMMIND_CONTEXT_ID: 

94 return 

95 data = event.data.get(_ATTR_SERVICE_DATA) or {} 

96 service = event.data.get(ATTR_SERVICE, "") 

97 eids = data.get("entity_id") 

98 for eid in [eids] if isinstance(eids, str) else list(eids or []): 

99 # Fired before the service runs, so the state is still the one the command changes 

100 state = hass.states.get(eid) 

101 self.note_command(eid, service, data, switch_to=_mode_switched_to(service, data, state and state.state)) 

102 

103 unsub: Callable[[], None] = hass.bus.async_listen(EVENT_CALL_SERVICE, _record, event_filter=_is_climate_call) 

104 return unsub 

105 

106 def note_command( 

107 self, entity_id: str, service: str, data: Mapping[str, Any], *, switch_to: str | None = None 

108 ) -> None: 

109 """Remember a command RoomMind sent to *entity_id*; *switch_to* is the mode it switches to.""" 

110 now = self._now() 

111 pending = self._switch.get(entity_id) 

112 # Resent every cycle while the device does not show it yet; the first send starts the wait 

113 if switch_to is not None and not (pending is not None and pending.mode == switch_to and not pending.shown): 

114 self._switch[entity_id] = _ModeSwitch(now, switch_to) 

115 if service != "set_temperature": 

116 return 

117 for attr in SETPOINT_ATTRS: 

118 value = _as_float(data.get(attr)) 

119 if value is not None: 

120 recent = [s for s in self._sent.get((entity_id, attr), []) if now - s[0] < DEVICE_SETPOINT_ECHO_SECONDS] 

121 self._sent[(entity_id, attr)] = [*recent, (now, value)] 

122 

123 def _settling(self, entity_id: str, hvac_state: str, now: float) -> bool: 

124 """True while a setpoint report may still be the device's answer to a mode switch RoomMind sent. 

125 

126 Until the device shows the commanded mode (it may report the restored setpoint 

127 first), and for a short while after it does. A mode someone else picked on the 

128 device does not count, nor does a later report in RoomMind's mode. 

129 """ 

130 last_state = self._state.get(entity_id) 

131 if last_state is None or last_state[0] != hvac_state: 

132 last_state = self._state[entity_id] = (hvac_state, now) 

133 switch = self._switch.get(entity_id) 

134 if switch is None or now - switch.at >= DEVICE_SETPOINT_SETTLE_SECONDS: 

135 return False 

136 if not switch.shown: 

137 if not _shows(switch.mode, hvac_state): 

138 return True 

139 switch.shown = True 

140 return _shows(switch.mode, hvac_state) and now - last_state[1] < DEVICE_SETPOINT_MODE_REPORT_SECONDS

142 def observe( 

143 self, 

144 entity_id: str, 

145 hvac_state: str, 

146 attributes: Mapping[str, Any], 

147 tolerance: float, 

148 echo_tolerance: float, 

149 ) -> dict[str, tuple[float, float]]: 

150 """Fold in the device's reported state; returns ``{attr: (before, after)}`` changed on the device. 

151 

152 Values are in the device's unit. *tolerance* is the smallest change that counts, 

153 *echo_tolerance* how far the device may round a value RoomMind sent. 

154 """ 

155 now = self._now() 

156 settling = self._settling(entity_id, hvac_state, now) 

157 changes: dict[str, tuple[float, float]] = {} 

158 for attr in SETPOINT_ATTRS: 

159 value = _as_float(attributes.get(attr)) 

160 if value is None: 

161 continue 

162 key = (entity_id, attr) 

163 last = self._seen.get(key) 

164 before = last[0] if last is not None and now - last[1] < DEVICE_SETPOINT_GAP_SECONDS else None 

165 self._seen[key] = (value, now) 

166 sent = [s for s in self._sent.get(key, []) if now - s[0] < DEVICE_SETPOINT_ECHO_SECONDS] 

167 echo = next((i for i in range(len(sent) - 1, -1, -1) if abs(value - sent[i][1]) <= echo_tolerance), None) 

168 # A shown send and everything sent before it can no longer explain a later value 

169 self._sent[key] = sent if echo is None else sent[echo + 1 :] 

170 if echo is not None or before is None or settling or abs(value - before) <= tolerance: 

171 continue 

172 changes[attr] = (before, value) 

173 return changes

175 def forget_readings(self, entity_id: str) -> None: 

176 """Drop the last readings of a device that went unavailable: it may come back with factory values.""" 

177 for attr in SETPOINT_ATTRS: 

178 self._seen.pop((entity_id, attr), None) 

179 

180 def prune(self, keep: set[str]) -> None: 

181 """Forget devices no room uses any more.""" 

182 self._switch = {eid: s for eid, s in self._switch.items() if eid in keep} 

183 self._state = {eid: s for eid, s in self._state.items() if eid in keep} 

184 self._sent = {key: s for key, s in self._sent.items() if key[0] in keep} 

185 self._seen = {key: v for key, v in self._seen.items() if key[0] in keep}
