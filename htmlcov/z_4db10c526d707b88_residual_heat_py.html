1"""Residual heat modeling for thermal mass (underfloor heating, radiators). 

2 

3After heating stops, the thermal mass (floor screed, radiator body) continues 

4releasing stored energy into the room. This module provides functions to 

5compute the decaying residual heat fraction so that the EKF and MPC can 

6account for it instead of mis-attributing the continued warming to solar 

7gain or insulation effects. 

8 

9The AC counterpart is the rebound charge: an AC cools the air, which drops 

10below the building mass and warms back up after the run (see thermal_model). 

11""" 

12

27def compute_residual_heat( 

28 elapsed_minutes: float, 

29 system_type: str, 

30 last_power_fraction: float = 1.0, 

31 heating_duration_minutes: float = 0.0, 

32 scale: float = 1.0, 

33 tau_minutes: float | None = None, 

34) -> float: 

35 """Compute normalised residual heat fraction after heating stopped. 

36 

37 The result is in [0, initial_fraction] and represents the fraction of 

38 ``beta_h`` (learned heating rate) still being delivered by thermal mass. 

39 

40 The *charge_fraction* accounts for how long heating was active before 

41 it stopped — a brief 10-minute run barely warms the screed, whereas 

42 a 4-hour run fully charges it. 

43 

44 Args: 

45 elapsed_minutes: Time since heating stopped (must be >= 0). 

46 system_type: Key into ``HEATING_SYSTEM_PROFILES`` (e.g. "underfloor"). 

47 last_power_fraction: Power fraction that was active before stopping. 

48 heating_duration_minutes: How long heating was active before stopping. 

49 scale: Learned per-room multiplier on the profile's ``initial_fraction`` 

50 (see ``ResidualHeatLearner``); 1.0 is the profile as is. 

51 tau_minutes: Learned per-room decay time of the afterglow; None uses the 

52 profile's ``tau_minutes``. The charge time stays the profile's. 

53 

54 Returns: 

55 Residual heat fraction in [0, initial_fraction * scale]. Returns 0.0 for unknown or empty 

56 system types (backwards-compatible default). 

57 """ 

58 profile = HEATING_SYSTEM_PROFILES.get(system_type) 

59 if not profile or elapsed_minutes < 0: 

60 return 0.0 

61 

62 initial = profile["initial_fraction"] 

63 tau_charge = profile.get("tau_charge_minutes", profile["tau_minutes"]) 

64 tau = profile["tau_minutes"] if tau_minutes is None else tau_minutes 

65 

66 if tau <= 0: 

67 return 0.0 

68 

69 # Charge fraction: how much energy the thermal mass absorbed 

70 if heating_duration_minutes > 0 and tau_charge > 0: 

71 charge_fraction = 1.0 - math.exp(-heating_duration_minutes / tau_charge) 

72 else: 

73 charge_fraction = 1.0 # assume fully charged if duration unknown 

74 

75 q = initial * scale * charge_fraction * math.exp(-elapsed_minutes / tau) * last_power_fraction 

76 return q if q >= RESIDUAL_HEAT_CUTOFF else 0.0

79def build_residual_series( 

80 elapsed_minutes: float, 

81 system_type: str, 

82 n_blocks: int, 

83 dt_minutes: float = 5.0, 

84 last_power_fraction: float = 1.0, 

85 heating_duration_minutes: float = 0.0, 

86 scale: float = 1.0, 

87 tau_minutes: float | None = None, 

88) -> list[float]: 

89 """Build a decaying residual heat series for MPC lookahead. 

90 

91 Each entry represents the q_residual at the start of the corresponding 

92 time block, evenly spaced by *dt_minutes*. 

93 

94 Returns: 

95 List of length *n_blocks* with decaying residual fractions. 

96 """ 

97 return [ 

98 compute_residual_heat( 

99 elapsed_minutes + i * dt_minutes, 

100 system_type, 

101 last_power_fraction, 

102 heating_duration_minutes, 

103 scale, 

104 tau_minutes, 

105 ) 

106 for i in range(n_blocks) 

107 ]

110def residual_tau(system_type: str, learned: float | None = None) -> float: 

111 """Decay time of *system_type*'s afterglow: the room's learned one, else the profile's (0.0 without one).""" 

112 if learned is not None: 

113 return learned 

114 profile = HEATING_SYSTEM_PROFILES.get(system_type) 

115 return float(profile["tau_minutes"]) if profile else 0.0

118def get_min_run_blocks(system_type: str, dt_minutes: float = 5.0) -> int: 

119 """Return the minimum number of MPC blocks for a given heating system. 

120 

121 Falls back to 2 blocks (the existing default) when the system type is 

122 unknown or empty. 

123 """ 

124 profile = HEATING_SYSTEM_PROFILES.get(system_type) 

125 if not profile or dt_minutes <= 0: 

126 return 2 # existing default 

127 return max(2, math.ceil(profile["min_run_minutes"] / dt_minutes))

130def heat_min_run_blocks( 

131 system_type: str, 

132 dt_minutes: float = 5.0, 

133 *, 

134 q_heat: float = 0.0, 

135 scale: float = 1.0, 

136 tau_minutes: float | None = None, 

137 fast_heater: bool = False, 

138) -> int: 

139 """Minimum heating run in blocks, shortened for radiators that store a lot of heat. 

140 

141 The rise the stored heat of one regular min run gives after the stop (learned 

142 ``scale`` and decay ``tau_minutes``, full power, losses left out) above 

143 ``RADIATOR_SHORT_RUN_AFTERGLOW_K`` makes the run a single block: the heat of the 

144 regular one alone would carry the room past its target. Not with a heat-capable AC 

145 in the room (``fast_heater``): ``q_heat`` is then the split's, and its compressor 

146 must not run 5-min cycles. 

147 """ 

148 base = get_min_run_blocks(system_type, dt_minutes) 

149 if fast_heater or system_type != HEATING_SYSTEM_RADIATOR or q_heat <= 0 or dt_minutes <= 0: 

150 return base 

151 rise = afterglow_rise_k(system_type, q_heat, base * dt_minutes, scale, tau_minutes) 

152 return 1 if rise > RADIATOR_SHORT_RUN_AFTERGLOW_K else base

155def afterglow_rise_k( 

156 system_type: str, 

157 q_heat: float, 

158 run_minutes: float, 

159 scale: float = 1.0, 

160 tau_minutes: float | None = None, 

161) -> float: 

162 """How far the stored heat of a full-power run of *run_minutes* lifts the room after the stop. 

163 

164 The whole afterglow (initial share x decay time), losses left out. 

165 """ 

166 q0 = compute_residual_heat(0.0, system_type, 1.0, run_minutes, scale, tau_minutes) 

167 return q_heat * q0 * residual_tau(system_type, tau_minutes) / 60.0

170def advance_rebound(charge: float, drive: float, dt_minutes: float) -> float: 

171 """AC rebound charge after *dt_minutes*. 

172 

173 The charge (0..1) is the air's offset below the building mass relative to a 

174 long run. It relaxes towards *drive* (1 while the AC cools, 0 otherwise) with 

175 ``AC_REBOUND_TAU_MINUTES``. On/off, not the power fraction: a unit runs near 

176 full power until its own sensor is satisfied, a low proportional setpoint 

177 does not change that. 

178 """ 

179 if dt_minutes <= 0: 

180 return charge 

181 result = drive + (charge - drive) * math.exp(-dt_minutes / AC_REBOUND_TAU_MINUTES) 

182 return result if result >= AC_REBOUND_CUTOFF else 0.0

185def mean_rebound(charge: float, drive: float, dt_minutes: float) -> float: 

186 """Mean charge over *dt_minutes* starting at *charge* (the ``q_rebound`` of that step).""" 

187 if dt_minutes <= 0: 

188 return charge 

189 tau = AC_REBOUND_TAU_MINUTES 

190 return drive + (charge - drive) * tau / dt_minutes * (1.0 - math.exp(-dt_minutes / tau))
