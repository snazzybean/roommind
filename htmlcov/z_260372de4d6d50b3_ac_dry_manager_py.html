1"""Air conditioners in dry while a room is too humid and needs neither heating nor cooling (#440). 

2 

3Rooms with an upper humidity limit and no dehumidifier of their own (or one unreadable for 

4``AC_DRY_DEHUMIDIFIER_GONE_MINUTES``) let an AC that offers ``dry`` run in it while the room is 

5too humid and the controller idles. Heating and cooling always come first: the room's mode is 

6decided before, and a room that heats or cools ends dry in the same cycle, ``async_apply`` then 

7switches the unit. Dry cools as well, so it keeps to the room's band: it ends 

8``AC_DRY_STOP_MARGIN_K`` above the heat target and starts again only ``AC_DRY_START_MARGIN_K`` 

9above it, and not within ``AC_DRY_AFTER_HEATING_MINUTES`` of the room heating. Like coil dry, a 

10running unit is left out of ``async_apply`` and counts as a running member of its compressor 

11group. 

12 

13The unit gets the stop point as its setpoint: most units dry like they cool, down to the 

14setpoint, so one left at its setback (cool 27 °C) would not dry at all, and the unit stops by 

15itself if RoomMind's room sensor lags. A unit idling in setback goes back to its mode and 

16setpoint afterwards; off and fan_only units get their idle action from ``async_apply``. 

17"""

59def dry_stop_temp(heat: float | None, cool: float | None, climate_mode: str) -> float | None: 

60 """Room temperature (°C) at which dry ends: above the heat target, at most 3 K under the cool one. 

61 

62 None without a target (the room is switched off). 

63 """ 

64 bounds: list[float] = [] 

65 if heat is not None and climate_mode != CLIMATE_MODE_COOL_ONLY: 

66 bounds.append(heat) 

67 if cool is not None: 

68 bounds.append(cool - AC_DRY_MAX_BELOW_COOL_K) 

69 return max(bounds) + AC_DRY_STOP_MARGIN_K if bounds else None

132 async def async_process_room( 

133 self, 

134 *, 

135 area_id: str, 

136 room: dict, 

137 mode: str, 

138 too_humid: bool, 

139 allowed: bool, 

140 may_cool: bool, 

141 control_active: bool, 

142 window_open: bool = False, 

143 dehumidifier_reads: bool | None = None, 

144 waiting: bool, 

145 room_temp: float | None, 

146 stop_temp: float | None, 

147 compressor_forced_on: set[str], 

148 blocked_eids: set[str], 

149 group_conflicts: dict[str, frozenset[str]], 

150 can_start: Callable[[str], bool], 

151 coil_dry_follows: Callable[[str], bool] = lambda _eid: False, 

152 switched_off: bool = False, 

153 ) -> AcDryRoomResult: 

154 """Advance dry for every AC of the room. 

155 

156 ``allowed``: the room may dry at all (Full Control, upper limit, switch on, not switched 

157 off, not heat only). ``window_open`` ends a run at once and gives no setback back: the 

158 window pause in async_apply takes the unit over in the same cycle. 

159 ``dehumidifier_reads``: None without a dehumidifier of the room's own; it is the tool the 

160 user picked, so an AC only starts once it was unreadable for 

161 ``AC_DRY_DEHUMIDIFIER_GONE_MINUTES``, and a run ends after its min run once it is back. 

162 ``may_cool``: the outdoor limit lets the room cool now; asked only for a start, a reading going back and forth over the cooling 

163 minimum must not end a run. ``coil_dry_follows``: units whose coil dry takes over after 

164 dry, they get no setback back. ``switched_off``: the room's schedule or presence switches 

165 it off, async_apply sends the off. ``control_active``: climate control on for the 

166 room; when off, a unit RoomMind runs in dry goes off once and nothing else is sent (#36). 

167 ``waiting``: no reading yet, nothing is sent and nothing changes. ``group_conflicts``: the 

168 modes each unit's uniform-mode group rules out for a running room. ``can_start`` asks the 

169 compressor group (min off, uniform mode) last, it may claim the group's way for the cycle. 

170 """ 

171 result = AcDryRoomResult() 

172 now = time.time() 

173 devices = room.get("devices", []) 

174 if mode == MODE_HEATING: 

175 self._heated_at[area_id] = now 

176 if not too_humid: 

177 self._declined.pop(area_id, None) 

178 if dehumidifier_reads is not False: 

179 self._dehumidifier_gone_since.pop(area_id, None) 

180 may_replace_dehumidifier = dehumidifier_reads is None or ( 

181 dehumidifier_reads is False 

182 and now - self._dehumidifier_gone_since.setdefault(area_id, now) >= AC_DRY_DEHUMIDIFIER_GONE_MINUTES * 60 

183 ) 

184 cold = room_temp is None or stop_temp is None or room_temp <= stop_temp 

185 warm_enough = ( 

186 room_temp is not None 

187 and stop_temp is not None 

188 and room_temp >= stop_temp - AC_DRY_STOP_MARGIN_K + AC_DRY_START_MARGIN_K 

189 ) 

190 heated_recently = now - self._heated_at.get(area_id, -math.inf) < AC_DRY_AFTER_HEATING_MINUTES * 60 

191 

192 for eid in get_ac_eids(devices): 

193 st = self._running.get(eid) 

194 if waiting: 

195 if st is not None: 

196 result.drying_eids.add(eid) 

197 continue 

198 state = self.hass.states.get(eid) 

199 if st is not None: 

200 st.area_id = area_id # a unit moved to another room goes on there 

201 reason = self._end_reason( 

202 eid, 

203 st, 

204 state, 

205 now, 

206 mode=mode, 

207 too_humid=too_humid, 

208 allowed=allowed, 

209 window_open=window_open, 

210 control_active=control_active, 

211 cold=cold, 

212 dehumidifier_back=dehumidifier_reads is True, 

213 held=eid in compressor_forced_on, 

214 conflicts=group_conflicts.get(eid, frozenset()), 

215 ) 

216 if reason is None: 

217 result.drying_eids.add(eid) 

218 await self._keep_setpoint(eid, st, state, area_id, stop_temp, devices) 

219 continue 

220 _LOGGER.debug("Area '%s': dry on '%s' ends (%s)", area_id, eid, reason) 

221 self._dried_at[area_id] = now 

222 if reason == _END_COLD: 

223 self._rest_until[area_id] = now + AC_DRY_MIN_OFF_MINUTES * 60 

224 restore = ( 

225 not switched_off 

226 and not coil_dry_follows(eid) 

227 and st.prev_mode not in group_conflicts.get(eid, frozenset()) 

228 ) 

229 if await self._async_end(eid, st, state, area_id, reason=reason, restore=restore): 

230 result.restored_eids.add(eid) 

231 continue 

232 if ( 

233 too_humid 

234 and allowed 

235 and not window_open 

236 and may_replace_dehumidifier 

237 and may_cool 

238 and control_active 

239 and mode == MODE_IDLE 

240 and warm_enough 

241 and not heated_recently 

242 and now >= self._rest_until.get(area_id, -math.inf) 

243 and eid not in self._declined.get(area_id, set()) 

244 and eid not in blocked_eids 

245 and eid not in compressor_forced_on 

246 and DRY not in group_conflicts.get(eid, frozenset()) 

247 and self._offers_dry(state) 

248 and can_start(eid) 

249 ): 

250 await self._async_start(eid, state, area_id, now, stop_temp, devices) 

251 result.drying_eids.add(eid) 

252 if result.drying_eids: 

253 self._dried_at[area_id] = now 

254 result.settling = now - self._dried_at.get(area_id, -math.inf) < AC_DRY_SETTLE_MINUTES * 60 

255 return result

257 @staticmethod 

258 def _offers_dry(state: Any) -> bool: 

259 if state is None or state.state in _NO_READING: 

260 return False 

261 return DRY in (state.attributes.get("hvac_modes") or []) 

262 

263 def _end_reason( 

264 self, 

265 eid: str, 

266 st: _Dry, 

267 state: Any, 

268 now: float, 

269 *, 

270 mode: str, 

271 too_humid: bool, 

272 allowed: bool, 

273 window_open: bool, 

274 control_active: bool, 

275 cold: bool, 

276 dehumidifier_back: bool, 

277 held: bool, 

278 conflicts: frozenset[str], 

279 ) -> str | None: 

280 """Why a running dry ends now, None while it goes on.""" 

281 if not control_active: 

282 return _END_CONTROL_OFF 

283 if mode != MODE_IDLE: 

284 return f"room {mode}" 

285 if window_open: 

286 return _END_WINDOW 

287 if not allowed: 

288 return _END_NOT_ALLOWED 

289 if DRY in conflicts: 

290 return "compressor group runs the other way" 

291 current = state.state if state is not None else None 

292 if current == DRY: 

293 st.confirmed = True 

294 elif current is not None and current not in _NO_READING: 

295 # Another mode: changed by hand, or the unit never took dry 

296 if st.confirmed or now - st.since >= AC_DRY_REPORT_SECONDS: 

297 self._declined.setdefault(st.area_id, set()).add(eid) 

298 return "changed on the device" if st.confirmed else "device does not take dry" 

299 return None 

300 if held or now - st.since < AC_DRY_MIN_RUN_MINUTES * 60: 

301 return None 

302 if not too_humid: 

303 return _END_DRY_ENOUGH 

304 if cold: 

305 return _END_COLD 

306 if dehumidifier_back: 

307 return _END_DEHUMIDIFIER_BACK 

308 return None

310 async def _async_start( 

311 self, eid: str, state: Any, area_id: str, now: float, stop_temp: float | None, devices: list[dict] 

312 ) -> None: 

313 idle_action, _ = get_idle_action(devices, eid) 

314 st = _Dry(area_id=area_id, since=now) 

315 if idle_action == IDLE_ACTION_SETBACK and state.state in ("heat", "cool"): 

316 temperature = state.attributes.get("temperature") 

317 if isinstance(temperature, int | float): 

318 st.prev_mode = state.state 

319 st.prev_temperature = float(temperature) 

320 self._running[eid] = st 

321 self._dirty = True 

322 _LOGGER.debug("Area '%s': '%s' dries (room too humid, no heating or cooling needed)", area_id, eid) 

323 if state.state != DRY: 

324 await async_climate_command(self.hass, eid, "set_hvac_mode", {"hvac_mode": DRY}, area_id=area_id) 

325 await self._keep_setpoint(eid, st, self.hass.states.get(eid), area_id, stop_temp, devices)

327 async def _keep_setpoint( 

328 self, eid: str, st: _Dry, state: Any, area_id: str, stop_temp: float | None, devices: list[dict] 

329 ) -> None: 

330 """The stop point as setpoint, sent again only when it moves (target change).""" 

331 if stop_temp is None or state is None or not isinstance(state.attributes.get("temperature"), int | float): 

332 return 

333 value = device_setpoint_ha(self.hass, state, stop_temp + get_setpoint_offset(devices, eid)) 

334 if st.setpoint is not None and abs(value - st.setpoint) < 0.05: 

335 return 

336 st.setpoint = value 

337 self._dirty = True 

338 if abs(float(state.attributes["temperature"]) - value) < 0.05: 

339 return 

340 await async_climate_command(self.hass, eid, "set_temperature", {"temperature": value}, area_id=area_id) 

341 

342 async def _async_end(self, eid: str, st: _Dry, state: Any, area_id: str, *, reason: str, restore: bool) -> bool: 

343 """Leave dry. True when the unit got its setback back here and stays out of async_apply this cycle. 

344 

345 Not when the room heats or cools (async_apply switches the unit) nor after a hand on it. 

346 """ 

347 del self._running[eid] 

348 self._dirty = True 

349 in_dry = state is not None and state.state == DRY 

350 if reason == _END_CONTROL_OFF: 

351 # #36: nothing is sent while control is off, only the dry RoomMind started goes off once 

352 if in_dry: 

353 await async_turn_off_climate(self.hass, eid, area_id=area_id) 

354 return False 

355 if ( 

356 restore 

357 and in_dry 

358 and reason in (_END_DRY_ENOUGH, _END_COLD, _END_NOT_ALLOWED, _END_DEHUMIDIFIER_BACK) 

359 and st.prev_mode is not None 

360 and st.prev_temperature is not None 

361 ): 

362 return await async_climate_command( 

363 self.hass, 

364 eid, 

365 "set_temperature", 

366 {"hvac_mode": st.prev_mode, "temperature": st.prev_temperature}, 

367 area_id=area_id, 

368 ) 

369 return False

394 """Units RoomMind ran in dry before a restart; they count as shown in dry (a hand since then ends it).""" 

395 self._running = {} 

396 if not isinstance(data, dict):
